OMP_NUM_THREADS=2 OPENBLAS_NUM_THREADS=1 python scripts/ionic_tdse_campaign.py \
  --auto-resume --device cpu --config configs/followup/ion_2p_counter.json \

python scripts/audit_followup_production.py --out-root "$HELIUM_OUTPUT_ROOT"

# 单个双电子/一电子任务也有入口。
